export type Ok<T> = { readonly ok: true; readonly value: T };
export type Err<E> = { readonly ok: false; readonly error: E };
export type Result<T, E> = Ok<T> | Err<E>;

export function ok<T>(value: T): Ok<T> {
  return { ok: true, value };
}

export function err<E>(error: E): Err<E> {
  return { ok: false, error };
}

export function isOk<T, E>(res: Result<T, E>): res is Ok<T> {
  return res.ok;
}

export function isErr<T, E>(res: Result<T, E>): res is Err<E> {
  return !res.ok;
}
Analyseur Lexical et Syntaxique : compiler/parser.tsTypeScript// compiler/parser.ts - Analyseur lexical et structures de jetons pour TokTS MVP
export type TokenType = 
  | "PUB" | "FN" | "TYPE" | "RET" | "USE" | "IF" | "ELSE"
  | "IDENT" | "STR_LIT" | "NUM_LIT" | "ARROW" | "ASSIGN"
  | "COLON" | "QUESTION" | "COMMA" | "LBRACE" | "RBRACE"
  | "LPAREN" | "RPAREN" | "LBRACKET" | "RBRACKET" | "DOT" | "EOF";

export interface Token {
  type: TokenType;
  value: string;
  line: number;
  col: number;
}

export function tokenize(source: string): Token[] {
  const tokens: Token[] = [];
  let pos = 0;
  let line = 1;
  let col = 1;

  while (pos < source.length) {
    const ch = source[pos];

    if (ch === " " || ch === "\t" || ch === "\r") {
      pos++;
      col++;
      continue;
    }
    if (ch === "\n") {
      line++;
      col = 1;
      pos++;
      continue;
    }
    if (ch === "/" && source[pos + 1] === "/") {
      while (pos < source.length && source[pos] !== "\n") {
        pos++;
      }
      continue;
    }

    if (source.startsWith("->", pos)) {
      tokens.push({ type: "ARROW", value: "->", line, col });
      pos += 2;
      col += 2;
      continue;
    }
    if (source.startsWith(":=", pos)) {
      tokens.push({ type: "ASSIGN", value: ":=", line, col });
      pos += 2;
      col += 2;
      continue;
    }

    const singleTokens: Record<string, TokenType> = {
      "{": "LBRACE", "}": "RBRACE", "(": "LPAREN", ")": "RPAREN",
      "[": "LBRACKET", "]": "RBRACKET", ":": "COLON", ",": "COMMA",
      "?": "QUESTION", ".": "DOT"
    };

    if (singleTokens[ch]) {
      tokens.push({ type: singleTokens[ch], value: ch, line, col });
      pos++;
      col++;
      continue;
    }

    if (ch === '"' || ch === "'") {
      const quote = ch;
      let val = "";
      pos++;
      col++;
      while (pos < source.length && source[pos] !== quote) {
        val += source[pos];
        pos++;
        col++;
      }
      pos++;
      col++;
      tokens.push({ type: "STR_LIT", value: val, line, col });
      continue;
    }

    if (/[0-9]/.test(ch)) {
      let numStr = "";
      while (pos < source.length && /[0-9.]/.test(source[pos])) {
        numStr += source[pos];
        pos++;
        col++;
      }
      tokens.push({ type: "NUM_LIT", value: numStr, line, col });
      continue;
    }

    if (/[a-zA-Z_]/.test(ch)) {
      let ident = "";
      while (pos < source.length && /[a-zA-Z0-9_\-\.\/]/.test(source[pos])) {
        ident += source[pos];
        pos++;
        col++;
      }
      const keywords: Record<string, TokenType> = {
        pub: "PUB", fn: "FN", type: "TYPE", ret: "RET",
        use: "USE", if: "IF", else: "ELSE"
      };
      const tokType = keywords[ident] || "IDENT";
      tokens.push({ type: tokType, value: ident, line, col });
      continue;
    }

    throw new Error(`Erreur lexicale : caractère inattendu '${ch}' à la position ${line}:${col}`);
  }

  tokens.push({ type: "EOF", value: "", line, col });
  return tokens;
}

import { tokenize, type Token } from "./parser.ts";

export interface CompileResult {
  ok: boolean;
  code: string;
  sourceMap?: string;
  error?: { formatted: string };
}

export function compileTokToTs(source: string, options: { filename?: string } = {}): CompileResult {
  try {
    const tokens = tokenize(source);
    let index = 0;

    function peek(): Token { return tokens[index]; }
    function next(): Token { return tokens[index++]; }
    function match(type: string): boolean {
      if (peek().type === type) { next(); return true; }
      return false;
    }
    function expect(type: string): Token {
      const t = next();
      if (t.type !== type) {
        throw new Error(`Attendu ${type} mais reçu ${t.type} ('${t.value}') à ${t.line}:${t.col}`);
      }
      return t;
    }

    let tsCode = "// Émis automatiquement par le compilateur TokTS\n";
    let needsResultRuntime = false;

    while (peek().type !== "EOF") {
      const tok = peek();

      if (tok.type === "USE") {
        next();
        const modPath = expect("IDENT").value;
        expect("LBRACE");
        const members: string[] = [];
        while (peek().type !== "RBRACE") {
          members.push(expect("IDENT").value);
          if (peek().type === "COMMA") next();
        }
        expect("RBRACE");
        tsCode += `import { ${members.join(", ")} } from "${modPath}";\n`;
        continue;
      }

      let isPub = false;
      if (peek().type === "PUB") {
        isPub = true;
        next();
      }

      if (peek().type === "TYPE") {
        next();
        const typeName = expect("IDENT").value;
        expect("COLON");
        const typeBody = parseTypeExpression();
        tsCode += `${isPub ? "export " : ""}type ${typeName} = ${typeBody};\n`;
        continue;
      }

      if (peek().type === "FN") {
        next();
        const fnName = expect("IDENT").value;
        expect("LPAREN");
        const params: string[] = [];
        while (peek().type !== "RPAREN") {
          const pName = expect("IDENT").value;
          expect("COLON");
          const pType = parseTypeExpression();
          params.push(`${pName}: ${pType}`);
          if (peek().type === "COMMA") next();
        }
        expect("RPAREN");

        let returnType = "void";
        if (peek().type === "COLON") {
          next();
          returnType = parseTypeExpression();
          if (returnType.startsWith("Result<")) {
            needsResultRuntime = true;
          }
        }

        expect("COLON");
        expect("LBRACE");
        const bodyContent = parseFunctionBody();
        expect("RBRACE");

        tsCode += `${isPub ? "export " : ""}function ${fnName}(${params.join(", ")}): ${returnType} {\n${bodyContent}\n}\n`;
        continue;
      }

      throw new Error(`Jeton inattendu à la racine : ${tok.value} (${tok.type}) à ${tok.line}:${tok.col}`);
    }

    if (needsResultRuntime && !tsCode.includes('from "./runtime/result.ts"')) {
      tsCode = 'import { type Result, ok, err } from "./runtime/result.ts";\n' + tsCode;
    }

    return { ok: true, code: tsCode };
  } catch (err: any) {
    return {
      ok: false,
      code: "",
      error: { formatted: err.message || String(err) }
    };
  }

  function parseTypeExpression(): string {
    const t = next();
    if (t.value === "str") return "string";
    if (t.value === "num") return "number";
    if (t.value === "bool") return "boolean";
    if (t.value === "nil") return "undefined";
    if (t.value === "any") return "any";

    if (t.type === "IDENT" && peek().type === "LBRACKET") {
      next();
      const args: string[] = [];
      while (peek().type !== "RBRACKET") {
        args.push(parseTypeExpression());
        if (peek().type === "COMMA") next();
      }
      expect("RBRACKET");
      return `${t.value}<${args.join(", ")}>`;
    }

    if (t.type === "LBRACE") {
      const fields: string[] = [];
      while (peek().type !== "RBRACE") {
        const fieldName = expect("IDENT").value;
        const isOptional = match("QUESTION");
        expect("COLON");
        const fieldType = parseTypeExpression();
        fields.push(`${fieldName}${isOptional ? "?" : ""}: ${fieldType}`);
        if (peek().type === "COMMA") next();
      }
      expect("RBRACE");
      return `{ ${fields.join("; ")} }`;
    }

    return t.value;
  }

  function parseFunctionBody(): string {
    const lines: string[] = [];
    let tmpCounter = 0;

    while (peek().type !== "RBRACE" && peek().type !== "EOF") {
      if (peek().type === "IDENT" && tokens[index + 1]?.type === "ASSIGN") {
        const varName = next().value;
        next();
        const expr = parseExpression();

        if (expr.endsWith("?")) {
          const callExpr = expr.slice(0, -1);
          const tempVar = `_res${tmpCounter++}`;
          lines.push(`  const ${tempVar} = ${callExpr};`);
          lines.push(`  if (!${tempVar}.ok) return ${tempVar};`);
          lines.push(`  const ${varName} = ${tempVar}.value;`);
        } else {
          lines.push(`  const ${varName} = ${expr};`);
        }
        continue;
      }

      if (peek().type === "IF") {
        next();
        const cond = parseExpression();
        expect("ARROW");
        const action = parseStatementOrExpression();
        lines.push(`  if (${cond}) {\n    ${action};\n  }`);
        continue;
      }

      if (peek().type === "RET") {
        next();
        const expr = parseExpression();
        lines.push(`  return ${expr};`);
        continue;
      }

      const raw = parseExpression();
      lines.push(`  ${raw};`);
    }

    return lines.join("\n");
  }

  function parseStatementOrExpression(): string {
    if (peek().type === "RET") {
      next();
      return `return ${parseExpression()}`;
    }
    return parseExpression();
  }

  function parseExpression(): string {
    let expr = "";
    while (
      peek().type !== "EOF" &&
      peek().type !== "RBRACE" &&
      peek().type !== "ARROW" &&
      peek().type !== "COMMA"
    ) {
      const t = next();
      if (t.type === "STR_LIT") expr += `"${t.value}"`;
      else expr += t.value;

      if (peek().type === "LPAREN" || peek().type === "DOT" || peek().type === "QUESTION") {
        continue;
      }
      expr += " ";
    }
    return expr.trim();
  }
}

import ts from "typescript";

export function compileTsToTok(tsCode: string): string {
  const sourceFile = ts.createSourceFile(
    "source.ts",
    tsCode,
    ts.ScriptTarget.ES2022,
    true,
    ts.ScriptKind.TS
  );

  let tokCode = "";

  function visit(node: ts.Node) {
    if (ts.isImportDeclaration(node)) {
      const modulePath = (node.moduleSpecifier as ts.StringLiteral).text;
      if (node.importClause?.namedBindings && ts.isNamedImports(node.importClause.namedBindings)) {
        const specifiers = node.importClause.namedBindings.elements.map(e => e.name.text);
        tokCode += `use ${modulePath} { ${specifiers.join(", ")} }\n`;
        return;
      }
    }

    if (ts.isTypeAliasDeclaration(node) || ts.isInterfaceDeclaration(node)) {
      const isPub = node.modifiers?.some(m => m.kind === ts.SyntaxKind.ExportKeyword);
      const name = node.name.text;
      tokCode += `${isPub ? "pub " : ""}type ${name} = `;

      if (ts.isInterfaceDeclaration(node)) {
        tokCode += formatMembers(node.members);
      } else {
        tokCode += formatTypeNode(node.type);
      }
      tokCode += "\n";
      return;
    }

    if (ts.isFunctionDeclaration(node) && node.name) {
      const isPub = node.modifiers?.some(m => m.kind === ts.SyntaxKind.ExportKeyword);
      const name = node.name.text;
      const params = node.parameters.map(p => {
        const pName = p.name.getText(sourceFile);
        const pType = p.type ? formatTypeNode(p.type) : "any";
        return `${pName}: ${pType}`;
      }).join(", ");

      const retType = node.type ? formatTypeNode(node.type) : "void";
      tokCode += `${isPub ? "pub " : ""}fn ${name}(${params}): ${retType} = {\n`;

      if (node.body) {
        node.body.statements.forEach(stmt => {
          tokCode += desugarStatement(stmt);
        });
      }
      tokCode += "}\n\n";
      return;
    }

    ts.forEachChild(node, visit);
  }

  function formatTypeNode(typeNode: ts.TypeNode): string {
    switch (typeNode.kind) {
      case ts.SyntaxKind.StringKeyword: return "str";
      case ts.SyntaxKind.NumberKeyword: return "num";
      case ts.SyntaxKind.BooleanKeyword: return "bool";
      case ts.SyntaxKind.UndefinedKeyword: return "nil";
      case ts.SyntaxKind.AnyKeyword: return "any";
    }

    if (ts.isTypeReferenceNode(typeNode)) {
      const typeName = typeNode.typeName.getText(sourceFile);
      if (typeNode.typeArguments && typeNode.typeArguments.length > 0) {
        const args = typeNode.typeArguments.map(formatTypeNode).join(", ");
        return `${typeName}[${args}]`;
      }
      return typeName;
    }

    if (ts.isTypeLiteralNode(typeNode)) {
      return formatMembers(typeNode.members);
    }

    return typeNode.getText(sourceFile);
  }

  function formatMembers(members: ts.NodeArray<ts.TypeElement>): string {
    const fields = members.map(m => {
      if (ts.isPropertySignature(m) && m.name) {
        const name = m.name.getText(sourceFile);
        const opt = m.questionToken ? "?" : "";
        const type = m.type ? formatTypeNode(m.type) : "any";
        return `${name}${opt}: ${type}`;
      }
      return "";
    }).filter(Boolean);
    return `{ ${fields.join(", ")} }`;
  }

  function desugarStatement(stmt: ts.Statement): string {
    if (ts.isReturnStatement(stmt)) {
      const expr = stmt.expression ? stmt.expression.getText(sourceFile) : "";
      return `  ret ${expr}\n`;
    }

    if (ts.isVariableStatement(stmt)) {
      const decl = stmt.declarationList.declarations[0];
      const name = decl.name.getText(sourceFile);
      const init = decl.initializer ? decl.initializer.getText(sourceFile) : "";
      return `  ${name} := ${init}\n`;
    }

    if (ts.isIfStatement(stmt)) {
      const cond = stmt.expression.getText(sourceFile);
      const thenText = stmt.thenStatement.getText(sourceFile).replace(/^{|}$/g, "").trim();
      return `  if ${cond} -> ${thenText}\n`;
    }

    return `  ${stmt.getText(sourceFile)}\n`;
  }

  visit(sourceFile);
  return tokCode.trim();
}
